%==============================================================================
\section{The standard basis of \texorpdfstring{$\R^{2}$}{R2}}
%==============================================================================
Since \S6 we have written a vector as a bracketed pair, $\vc{v}=[\,v_1,\,v_2\,]$,
and treated those two numbers as simply given. This section asks where they come
from. The answer is that the bracket notation has been making a silent choice
all along --- and once we see what the choice is, we can start asking what else
could have been chosen.

%------------------------------------------------------------------------------
\subsection{Two special vectors}
%------------------------------------------------------------------------------
\begin{defn}[The standard basis vectors]
The \textbf{standard basis vectors} of $\R^{2}$ are
\[
  \ivec = [\,1,\,0\,],
  \qquad
  \jvec = [\,0,\,1\,] .
\]
They are written with arrows overhead, one of the three conventions of \S6.
\end{defn}

\noindent Geometrically $\ivec$ is one step east and $\jvec$ is one step north:
the two elementary moves, one along each axis.

\begin{center}
\begin{tikzpicture}[scale=1.15]
  \lattice{-1}{3}{-1}{3}
  \draw[vecarr,RoyalBlue]   (0,0) -- (1,0);
  \draw[vecarr,ForestGreen] (0,0) -- (0,1);
  \draw[gray!75] (0.24,0) -- (0.24,0.24) -- (0,0.24);
  \node[below right,font=\small,RoyalBlue]  at (1.0,-0.04) {$\ivec=[1,0]$};
  \node[above left,font=\small,ForestGreen] at (0.02,1.05) {$\jvec=[0,1]$};
\end{tikzpicture}
\end{center}

\noindent Three properties, each of which we already have the tools to check.

\begin{itemize}[leftmargin=1.4em,itemsep=4pt]
  \item \textbf{Both have unit length.}
        $\norm{\ivec}=\sqrt{1^{2}+0^{2}}=1$ and
        $\norm{\jvec}=\sqrt{0^{2}+1^{2}}=1$, so both lie on the unit circle
        $\Sph{1}$ of \S12.1.
  \item \textbf{They are orthogonal.} Their dot product is
        $\langle\ivec,\jvec\rangle = 1\cdot 0 + 0\cdot 1 = 0$, so by the
        orthogonality test of \S11.7 they meet at a right angle --- as the
        figure shows, since they lie along the two axes.
  \item \textbf{Neither is a multiple of the other.} No scalar $\alpha$
        satisfies $\alpha[1,0]=[0,1]$, since the first coordinate would force
        $\alpha=0$ and the second would force $\alpha\cdot0=1$. So by \S10.1
        they are not parallel. Neither one is redundant: you cannot manufacture
        $\jvec$ out of $\ivec$ alone.
\end{itemize}

\noindent A collection of vectors that are mutually orthogonal and each of unit
length is called \textbf{orthonormal} --- ``ortho'' for the right angles,
``normal'' for the unit lengths. So $\ivec$ and $\jvec$ form an orthonormal
pair, which is as convenient an arrangement as two vectors can have. \S16.4
shows what that convenience buys.

\begin{warn}
This course uses the letters $i$, $j$ and $k$ for three different things, and
you will meet all of them. In the complex numbers lecture, $i$ is the
\emph{imaginary unit} with $i^{2}=-1$, and $i,j,k$ are the \emph{basis elements}
of the quaternions. Here $\ivec$ and $\jvec$ are \emph{vectors} in $\R^{2}$. The
arrows are what tell them apart, so keep them on.
\end{warn}

%------------------------------------------------------------------------------
\subsection{Building vectors out of them}
%------------------------------------------------------------------------------
Take any two real numbers $\alpha$ and $\beta$ and form
\[
  \alpha\ivec + \beta\jvec .
\]
Nothing here is new: it is scalar multiplication from \S8 followed by addition
from \S7. Working it out,
\[
  \alpha\ivec + \beta\jvec
  = \alpha[\,1,\,0\,] + \beta[\,0,\,1\,]
  = [\,\alpha,\,0\,] + [\,0,\,\beta\,]
  = [\,\alpha,\,\beta\,] ,
\]
which is a vector in $\R^{2}$. So every such combination lands inside $\R^{2}$.

\begin{defn}[Linear combination]
An expression of the form $\alpha\vc{u}+\beta\vc{v}$, with $\alpha,\beta\in\R$,
is called a \textbf{linear combination} of $\vc{u}$ and $\vc{v}$.
\end{defn}

\noindent The interesting direction is the reverse one. Does every vector of
$\R^{2}$ arise this way, or do these combinations only reach part of the plane?
They reach all of it, and the proof is a single line: given any
$\vc{v}=[\,v_1,\,v_2\,]$, choose $\alpha=v_1$ and $\beta=v_2$, and
\[
  v_1\ivec + v_2\jvec = [\,v_1,\,v_2\,] = \vc{v} .
\]
Nothing in $\R^{2}$ is missed. Two vectors, scaled and added, generate the
entire plane --- which is why $\ivec$ and $\jvec$ deserve to be called the
\emph{scaffolding} of $\R^{2}$: build them once and everything else follows.

%------------------------------------------------------------------------------
\subsection{Span and basis}
%------------------------------------------------------------------------------
Those two paragraphs proved two inclusions, and together they deserve a name.

\begin{defn}[Span]
The \textbf{span} of a collection of vectors is the set of \emph{all} their
linear combinations. For $\ivec$ and $\jvec$,
\[
  \spanop\{\ivec,\jvec\}
  \;=\;
  \bigl\{\, \alpha\ivec + \beta\jvec \;\bigm|\; \alpha,\beta\in\R \,\bigr\}.
\]
\end{defn}

\noindent The first calculation of \S16.2 showed
$\spanop\{\ivec,\jvec\}\subseteq\R^{2}$, and the second showed
$\R^{2}\subseteq\spanop\{\ivec,\jvec\}$. Both inclusions hold, so
\[
  \boxed{\;\spanop\{\ivec,\jvec\} \;=\; \R^{2}\;}
\]
and we say that $\ivec$ and $\jvec$ \textbf{span} $\R^{2}$.

\begin{defn}[Basis, working definition]
A collection of vectors is a \textbf{basis} for a space when it spans that space
and does so without redundancy --- no vector in the collection is a linear
combination of the others, so none of them can be dropped. We write
\[
  \bas = \{\,\ivec,\ \jvec\,\} ,
\]
and call $\bas$ the \textbf{standard basis} of $\R^{2}$.
\end{defn}

\begin{rmk}[Basis and span are different objects]
Keep the two apart. The basis $\bas=\{\ivec,\jvec\}$ is a set containing
\emph{two vectors}. Its span, $\spanop\bas=\R^{2}$, is a set containing
\emph{infinitely many} --- every vector in the plane. A basis is small; what it
generates is large. That is the entire point of a basis: it is a finite
description of an infinite set.
\end{rmk}

\noindent Redundancy is the clause worth dwelling on. Spanning alone is easy to
achieve by piling on extra vectors: the collection
$\{\ivec,\jvec,[\,3,\,5\,]\}$ also spans $\R^{2}$, but the third vector earns
nothing, since $[\,3,\,5\,]=3\ivec+5\jvec$ was already reachable. A basis is
required to carry no such passengers, and \S16.1 checked that ours does not:
neither $\ivec$ nor $\jvec$ is a multiple of the other.

\lookahead ``Without redundancy'' is doing real work here and deserves a
sharper statement than the one above. It has a proper name --- \emph{linear
independence} --- and a definition that applies to any number of vectors at
once, not just two. We will meet it shortly.

%------------------------------------------------------------------------------
\subsection{Coordinates}
%------------------------------------------------------------------------------
\begin{defn}[Coordinates]
If $\vc{v}=\alpha\ivec+\beta\jvec$, the scalars $\alpha$ and $\beta$ are the
\textbf{coordinates} of $\vc{v}$ with respect to the basis $\bas$.
\end{defn}

\noindent For the word ``the'' to be justified, there had better be only one
such pair --- and there is. Suppose a vector could be written two ways,
\[
  \alpha\ivec + \beta\jvec = \alpha'\ivec + \beta'\jvec .
\]
Both sides evaluate componentwise to $[\,\alpha,\,\beta\,]$ and
$[\,\alpha',\,\beta'\,]$, and two vectors are equal exactly when their
coordinates agree (\S9). So $\alpha=\alpha'$ and $\beta=\beta'$: the two
representations were the same one. Coordinates are \textbf{unique}.

\begin{rmk}[What the bracket notation was hiding]
Look again at $\vc{v}=[\,v_1,\,v_2\,]$. We have just shown
$\vc{v}=v_1\ivec+v_2\jvec$, so the two numbers inside the brackets are exactly
the coordinates of $\vc{v}$ with respect to $\bas$. The notation of \S6 was
never neutral --- it silently reported coordinates in one particular basis. We
have been able to ignore this only because $\bas$ has been the sole basis in
play. Other bases exist, and the same vector has different coordinates in each.
\end{rmk}

\begin{rmk}[Why reading them off is free]
There is a reason the coordinates turned out to be the components themselves.
Because $\bas$ is orthonormal, the projection machinery of \S12.4 collapses:
\[
  \comp{\ivec}{\vc{v}}
  = \frac{\langle\ivec,\vc{v}\rangle}{\norm{\ivec}}
  = \frac{1\cdot v_1 + 0\cdot v_2}{1}
  = v_1 ,
\]
and likewise $\comp{\jvec}{\vc{v}}=v_2$. So the coordinate of $\vc{v}$ along a
basis vector \emph{is} its projection component onto that basis vector --- the
shadow it casts on each axis. Orthonormality is what makes the norms $1$ and the
cross terms $0$, and so makes the arithmetic vanish.
\end{rmk}

%------------------------------------------------------------------------------
\subsection{The reverse problem, and a first linear system}
%------------------------------------------------------------------------------
Turn the question around. Instead of being handed $\alpha$ and $\beta$ and
asked what they build, start with a vector $\vc{v}\in\R^{2}$ and treat the
scalars as \emph{unknowns}:
\[
  \vc{v} = \alpha\ivec + \beta\jvec ,
  \qquad \alpha,\beta \text{ to be found.}
\]
Two facts already settle that this is a sensible question. Because $\ivec$ and
$\jvec$ span $\R^{2}$, a solution \emph{exists} --- no vector of the plane is
out of reach. Because coordinates are unique, the solution is the \emph{only}
one. We are not searching hopefully; we are computing something we know is there
and know is unambiguous.

\subsubsection*{The picture}
Geometrically the answer is a walk. Starting at the origin, travel $\alpha$
steps along $\ivec$ --- purely horizontal --- and then, from where you land,
$\beta$ steps along $\jvec$ --- purely vertical. That is the head-to-tail
reading of addition from \S7.1, and the question is how far to walk in each
direction to arrive exactly at the head of $\vc{v}$.

Read the other way, the dashed guides drop from the head of $\vc{v}$ onto the
two axes, and where they land is the answer.

\begin{center}
\begin{tikzpicture}[scale=1.15]
  \lattice{-1}{4}{-1}{3}
  \draw[gray!55,dashed] (3,2) -- (3,0);
  \draw[gray!55,dashed] (3,2) -- (0,2);
  \draw[vecarr,RoyalBlue]   (0,0) -- (3,0);
  \draw[vecarr,ForestGreen] (3,0) -- (3,2);
  \draw[vecarr,BrickRed]    (0,0) -- (3,2);
  \node[below,font=\scriptsize,RoyalBlue]      at (1.5,-0.06) {$\alpha\ivec=3\ivec$};
  \node[right,font=\scriptsize,ForestGreen]    at (3.06,1.0)  {$\beta\jvec=2\jvec$};
  \node[above left,font=\small,BrickRed]       at (2.05,1.5)  {$\vc{v}=[3,2]$};
  \node[left,font=\scriptsize,gray]            at (-0.06,2)   {$2$};
  \node[below,font=\scriptsize,gray]           at (3,-0.5)    {$3$};
\end{tikzpicture}
\end{center}

\subsubsection*{Writing it as a system}
Now do it algebraically. Expand the right-hand side componentwise:
\[
  \alpha\ivec + \beta\jvec
  = [\,\alpha\cdot 1 + \beta\cdot 0,\;\; \alpha\cdot 0 + \beta\cdot 1\,] .
\]
Setting this equal to $\vc{v}=[\,v_1,\,v_2\,]$ and matching coordinates gives
two equations in the two unknowns:
\[
  \begin{aligned}
    v_1 &= \alpha\cdot 1 + \beta\cdot 0, \\
    v_2 &= \alpha\cdot 0 + \beta\cdot 1 .
  \end{aligned}
\]
A collection of equations like this --- several linear equations sharing the
same unknowns, to be satisfied simultaneously --- is called a \textbf{linear
system}. This one barely deserves the name, because the zeros knock out one term
in each equation and it collapses on sight:
\[
  v_1 = \alpha,
  \qquad
  v_2 = \beta .
\]

\begin{ex}
For $\vc{v}=[\,3,\,2\,]$ the system gives $\alpha=3$ and $\beta=2$, so
$\vc{v}=3\ivec+2\jvec$ --- the walk in the figure above. For
$\vc{v}=[\,-2,\,5\,]$ it gives $\vc{v}=-2\ivec+5\jvec$: two steps west, then
five north.
\end{ex}

\begin{rmk}
Do not mistake how easy that was for how easy the question is. The system
collapsed because the coefficient of $\beta$ in the first equation and of
$\alpha$ in the second were both $0$, and those zeros are precisely the
statement $\langle\ivec,\jvec\rangle=0$ --- the orthogonality we checked in
\S16.1, reappearing as the arithmetic that makes the work disappear. The
standard basis is engineered so that reading coordinates costs nothing.
\end{rmk}

\lookahead Ask the same question against a basis that is \emph{not} the standard
one and the zeros are gone. The two equations then genuinely interact, both
unknowns appear in both, and the system must actually be solved. That is the
general problem of linear systems --- how to state it, when it has a solution,
how many solutions there can be, and how to compute them --- and it will occupy
a substantial part of this course. The trivial case above is worth remembering
as the one you are always trying to get back to.

\section{Dimension}
%==============================================================================
A basis carries a number with it, and that number turns out to be one we have
been using informally since \S1.

\begin{defn}[Dimension]
The \textbf{dimension} of a space is the number of vectors in a basis for it.
\end{defn}

\noindent Our basis $\bas=\{\ivec,\jvec\}$ contains two vectors, so
\[
  \dim\R^{2} = 2 .
\]
The superscript in $\R^{2}$ was telling us the dimension all along.

\subsection*{Why this agrees with what we already meant}
In \S1 we described dimension as a \emph{degree of freedom} with respect to
spatial coordinates: how many independent numbers must you supply to pin down a
position? The two accounts are the same account. Supplying a position in
$\R^{2}$ means supplying its two coordinates $\alpha$ and $\beta$ --- one per
basis vector --- so the count of free numbers and the count of basis vectors are
forced to agree.

Read geometrically, the dimension counts \textbf{independent directions}. The
basis $\bas$ offers two: along the $x$-axis and along the $y$-axis. Neither can
be obtained from the other, and once both are available, nothing further is
needed.

\begin{warn}
``Two independent directions'' does not mean the basis represents only two
directions. It represents \emph{every} direction in the plane --- the vector
$3\ivec+2\jvec$ points in a direction that is neither of the two axes. What the
number $2$ counts is how many directions you must be \emph{given} before all the
rest come free. Two is the number of directions that are independent, not the
number that are reachable.
\end{warn}

\section{Other bases, and linear independence}
%==============================================================================
Nothing in \S16.3 said the standard basis is the \emph{only} basis for $\R^{2}$.
It is merely the most convenient one. Once we go looking for others, a question
appears that the standard basis was too well behaved to raise: which pairs of
vectors work, and which fail?

%------------------------------------------------------------------------------
\subsection{Bases other than the standard one}
%------------------------------------------------------------------------------
Consider the pairs
\[
  \vc{u}=[\,1,\,2\,],\ \vc{v}=[\,2,\,3\,]
  \qquad\text{and}\qquad
  \vc{u}=[\,2,\,3\,],\ \vc{v}=[\,3,\,4\,] .
\]
Neither pair looks anything like $\{\ivec,\jvec\}$ --- the vectors are not unit
length, not orthogonal, not aligned with the axes --- and yet each pair is a
perfectly good basis for $\R^{2}$.

\begin{center}
\begin{tikzpicture}[scale=0.85,baseline=(current bounding box.center)]
  \lattice{-1}{4}{-1}{5}
  \draw[vecarr,RoyalBlue]   (0,0) -- (1,2);
  \draw[vecarr,ForestGreen] (0,0) -- (2,3);
  \node[left,font=\scriptsize,RoyalBlue]    at (0.94,2.06) {$\vc{u}=[1,2]$};
  \node[right,font=\scriptsize,ForestGreen] at (2.06,3.02) {$\vc{v}=[2,3]$};
\end{tikzpicture}
\hspace{1.1cm}
\begin{tikzpicture}[scale=0.85,baseline=(current bounding box.center)]
  \lattice{-1}{4}{-1}{5}
  \draw[vecarr,RoyalBlue]   (0,0) -- (2,3);
  \draw[vecarr,ForestGreen] (0,0) -- (3,4);
  \node[left,font=\scriptsize,RoyalBlue]    at (1.94,3.06) {$\vc{u}=[2,3]$};
  \node[right,font=\scriptsize,ForestGreen] at (3.06,4.02) {$\vc{v}=[3,4]$};
\end{tikzpicture}
\end{center}

\noindent Being a basis costs something, though: the coordinates are no longer
free to read off.

\begin{ex}[Coordinates against a non-standard basis]
Take the basis $\{\vc{u},\vc{v}\}$ with $\vc{u}=[1,2]$ and $\vc{v}=[2,3]$, and
ask for the coordinates of $\vc{w}=[\,1,\,0\,]$. We need $\alpha,\beta$ with
$\alpha\vc{u}+\beta\vc{v}=\vc{w}$, which componentwise is
\[
  \begin{aligned}
    \alpha\cdot 1 + \beta\cdot 2 &= 1, \\
    \alpha\cdot 2 + \beta\cdot 3 &= 0 .
  \end{aligned}
\]
No zeros this time, so nothing collapses. The second equation gives
$\alpha=-\tfrac32\beta$; substituting into the first,
$-\tfrac32\beta+2\beta=\tfrac12\beta=1$, so $\beta=2$ and $\alpha=-3$. Checking:
\[
  -3[\,1,\,2\,] + 2[\,2,\,3\,] = [\,-3,\,-6\,]+[\,4,\,6\,] = [\,1,\,0\,] .
  \quad\checkmark
\]
So $\vc{w}$ has coordinates $-3$ and $2$ in this basis, where in the standard
basis it had coordinates $1$ and $0$. Same vector, different numbers --- exactly
what the remark in \S16.4 warned. And this is the situation \S16.5 pointed at:
away from the standard basis, finding coordinates is a genuine linear system
that has to be solved.
\end{ex}

%------------------------------------------------------------------------------
\subsection{When two vectors do not span the plane}
%------------------------------------------------------------------------------
Now try
\[
  \vc{u}=[\,1,\,2\,],
  \qquad
  \vc{v}=[\,2,\,4\,] .
\]
Plot them and something is visibly wrong: both arrows lie along one and the same
line through the origin.

\begin{center}
\begin{tikzpicture}[scale=0.85]
  \lattice{-3}{4}{-3}{5}
  % the line spanned by u -- every combination of u and v lands on it
  \draw[gray!65,dotted,thick] (-1.55,-3.1) -- (2.55,5.1);
  \draw[vecarr,ForestGreen] (0,0) -- (2,4);
  \draw[vecarr,RoyalBlue]   (0,0) -- (1,2);
  % keep the labels clear of the line the two vectors sit on
  \node[left,font=\scriptsize,RoyalBlue]   at (0.72,2.10) {$\vc{u}=[1,2]$};
  \node[left,font=\scriptsize,ForestGreen] at (1.72,4.20) {$\vc{v}=[2,4]$};
  \node[right,font=\scriptsize,gray]       at (2.5,4.92)  {$y=2x$};
\end{tikzpicture}
\end{center}

\noindent The reason is immediate: $\vc{v}=2\vc{u}$. And that single fact
collapses every linear combination onto the same line:
\[
  \alpha\vc{u} + \beta\vc{v}
  = \alpha\vc{u} + \beta(2\vc{u})
  = (\alpha+2\beta)\,\vc{u} ,
\]
which is a scalar multiple of $\vc{u}$ --- and by \S8 every scalar multiple of
$\vc{u}$ lies on the line through the origin and the head of $\vc{u}$. Whatever
$\alpha$ and $\beta$ you choose, you never leave that line.

So there is no way to assign coordinates to a point off the line: the point is
simply not reachable. These two vectors span only a \emph{subset} of the plane,
and therefore
\[
  \spanop\{\vc{u},\vc{v}\} \neq \R^{2}
  \qquad\text{when}\qquad
  \vc{u}=[1,2],\ \vc{v}=[2,4].
\]
Two vectors are not automatically a basis for $\R^{2}$. Having the right
\emph{number} of vectors is not enough; they have to stand in the right
relationship to one another.

%------------------------------------------------------------------------------
\subsection{Three questions}
%------------------------------------------------------------------------------
That failure is worth more than the successes, because it tells us there is
something to characterise. Three questions follow, and we take them one at a
time.

\begin{enumerate}[label=\textbf{Q\arabic*.},leftmargin=3.0em,itemsep=5pt,
                  topsep=5pt]
  \item Which pairs of vectors form a basis for $\R^{2}$ --- that is, which
        pairs span $\R^{2}$?
  \item The pair $[\,1,\,2\,]$ and $[\,2,\,4\,]$ failed to span $\R^{2}$, but
        they clearly span \emph{something}. What kind of basis do they form?
  \item How many different bases does $\R^{2}$ have?
\end{enumerate}

%------------------------------------------------------------------------------
\subsection{Question 1: linear independence}
%------------------------------------------------------------------------------
Q1 is better asked as a question about a \emph{relationship}: what must be true
of two vectors, with respect to each other, for them to form a basis?

The two examples answer it geometrically. The vectors $[2,3]$ and $[3,4]$ point
in genuinely different directions, and no amount of stretching turns one into
the other: there is no scalar $\alpha$ with $\alpha[2,3]=[3,4]$, since the first
coordinate demands $\alpha=\tfrac32$ and the second then gives
$\tfrac32\cdot3=\tfrac92\neq4$. The pair $[1,2]$ and $[2,4]$ fails exactly here:
$2\cdot[1,2]=[2,4]$, so one \emph{is} a stretch of the other, and that is what
flattened their span onto a line.

So the property we want is ``neither vector is a scalar multiple of the other''
--- which is the \emph{not parallel} of \S10.1. That phrasing is correct, but it
is awkward: it treats the two vectors asymmetrically, and it does not generalise
past two. The standard formulation fixes both problems by asking instead about
combinations that produce the zero vector:
\[
  \alpha[\,2,\,3\,] + \beta[\,3,\,4\,] = \vc{0}
  \qquad\Longleftrightarrow\qquad
  \alpha=\beta=0 .
\]
Read the left-to-right direction as the real content: the \emph{only} way to
combine these two vectors and land back at the origin is to take nothing of
either.

\begin{defn}[Linear independence and dependence in $\R^{2}$]
Two vectors $\vc{u},\vc{v}\in\R^{2}$ are \textbf{linearly independent} when the
only scalars $\alpha,\beta\in\R$ satisfying
\[
  \alpha\vc{u} + \beta\vc{v} = \vc{0}
\]
are $\alpha=0$ and $\beta=0$.

They are \textbf{linearly dependent} when they are not --- that is, when there
exist scalars $\alpha,\beta$, \emph{not both zero}, with
$\alpha\vc{u}+\beta\vc{v}=\vc{0}$.
\end{defn}

\begin{rmk}[Why ``not both zero'' cannot be dropped]
The choice $\alpha=\beta=0$ always works, for any two vectors whatsoever, since
$0\vc{u}+0\vc{v}=\vc{0}$ is automatic. It carries no information. So a
definition of dependence reading ``there exist $\alpha,\beta$ with
$\alpha\vc{u}+\beta\vc{v}=\vc{0}$'' would be satisfied by every pair and would
say nothing at all. The clause \emph{not both zero} is what gives the definition
teeth --- and for the same reason, independence is stated as
$\alpha=\beta=0$ being the \emph{only} solution.
\end{rmk}

\begin{ex}[The two pairs, checked]
For $\vc{u}=[1,2]$ and $\vc{v}=[2,4]$, take $\alpha=2$ and $\beta=-1$:
\[
  2[\,1,\,2\,] + (-1)[\,2,\,4\,] = [\,2,\,4\,] - [\,2,\,4\,] = \vc{0} ,
\]
and those scalars are not both zero. So the pair is linearly
\textbf{dependent} --- and we have already seen the consequence: their span is a
line, not the plane.

For $\vc{u}=[2,3]$ and $\vc{v}=[3,4]$, suppose
$\alpha[2,3]+\beta[3,4]=\vc{0}$. Componentwise that is $2\alpha+3\beta=0$ and
$3\alpha+4\beta=0$. The first gives $\alpha=-\tfrac32\beta$; substituting into
the second, $-\tfrac92\beta+4\beta=-\tfrac12\beta=0$, so $\beta=0$ and then
$\alpha=0$. The only solution is the trivial one, so the pair is linearly
\textbf{independent}.
\end{ex}

\begin{rmk}[For two vectors, dependence is exactly parallelism]
When $\vc{u}$ and $\vc{v}$ are both nonzero, linear dependence says nothing more
than what \S10.1 called parallel. If $\alpha\vc{u}+\beta\vc{v}=\vc{0}$ with, say,
$\alpha\neq0$, then $\vc{u}=-\tfrac{\beta}{\alpha}\vc{v}$, so each is a scalar
multiple of the other; and conversely a multiple relation rearranges into such a
combination. The one wrinkle is the zero vector: $\{\vc{0},\vc{v}\}$ is always
dependent, since $1\cdot\vc{0}+0\cdot\vc{v}=\vc{0}$ uses a nonzero scalar --- yet
\S10.1 deliberately excluded $\vc{0}$ from the word ``parallel'' because it has
no direction. The combination formulation handles that case without special
pleading, which is one more reason it is the definition of record.
\end{rmk}

\noindent This answers Q1, at least for pairs: \textbf{two vectors form a basis
for $\R^{2}$ precisely when they are linearly independent.} Independence is what
guarantees the two arrows genuinely open the plane out, instead of collapsing it
onto a line.

\lookahead The definition above was written for two vectors so that it can be
checked by hand, but nothing in it depends on there being two. The general
version asks the same question of any collection --- when is the zero vector
reachable only by taking nothing of everything? --- and that is the version we
will need once we leave the plane. Q2 and Q3 remain open.

%------------------------------------------------------------------------------
\subsection{Independent vectors really do span the plane}
%------------------------------------------------------------------------------
The last line of \S18.4 was an assertion, not a proof. We claimed that two
linearly independent vectors form a basis for $\R^{2}$, on the strength of two
examples and a picture. It is worth proving, because the proof is short, because
it produces a formula we can use, and because it is the first time in this unit
that independence does genuine work rather than merely being defined.

\begin{quote}
\textbf{Claim.} Let $\vc{u},\vc{v}\in\R^{2}$ be linearly independent. Then
\[
  \spanop\{\vc{u},\vc{v}\} = \R^{2} .
\]
\end{quote}

\noindent Two sets are equal when each contains the other, so there are two
things to show. Write $\vc{u}=[\,u_1,\,u_2\,]$ and $\vc{v}=[\,v_1,\,v_2\,]$
throughout.

\subsubsection*{Step 1: every combination lands in \texorpdfstring{$\R^{2}$}{R2}}
This direction needs no hypothesis at all. For any $\alpha,\beta\in\R$,
\[
  \alpha\vc{u} + \beta\vc{v}
  = [\,\alpha u_1 + \beta v_1,\;\; \alpha u_2 + \beta v_2\,] .
\]
Each entry is built from real numbers by multiplication and addition, so each
entry is a real number, so the result is an ordered pair of reals --- an element
of $\R^{2}$ by the definition in \S6. Hence
\[
  \spanop\{\vc{u},\vc{v}\} \subseteq \R^{2} .
\]

\subsubsection*{Step 2: every element of \texorpdfstring{$\R^{2}$}{R2} is such a combination}
This is the direction that needs independence. Let $\vc{w}=[\,w_1,\,w_2\,]$ be
any vector in $\R^{2}$. We must produce scalars $\alpha,\beta$ with
$\alpha\vc{u}+\beta\vc{v}=\vc{w}$, which componentwise is the linear system
\begin{equation}
  \begin{aligned}
    \alpha u_1 + \beta v_1 &= w_1, \\
    \alpha u_2 + \beta v_2 &= w_2 .
  \end{aligned}
  \label{eq:span}
\end{equation}
Eliminate one unknown at a time. Multiplying the first equation by $v_2$, the
second by $v_1$, and subtracting, the $\beta$ terms cancel and leave
\[
  \alpha\,(u_1v_2 - u_2v_1) = w_1v_2 - w_2v_1 .
\]
Multiplying the first by $u_2$, the second by $u_1$, and subtracting, the
$\alpha$ terms cancel and leave
\[
  \beta\,(u_2v_1 - u_1v_2) = u_2w_1 - u_1w_2,
  \qquad\text{that is}\qquad
  \beta\,(u_1v_2 - u_2v_1) = u_1w_2 - u_2w_1 .
\]
The same quantity governs both, so give it a name:
\[
  D = u_1v_2 - u_2v_1 .
\]
If $D\neq0$ we may divide, and the system is solved outright:
\begin{equation}
  \alpha = \frac{w_1v_2 - w_2v_1}{D},
  \qquad
  \beta  = \frac{u_1w_2 - u_2w_1}{D} .
  \label{eq:coords}
\end{equation}
So everything turns on whether $D$ can be zero. That is exactly what
independence rules out.

\subsubsection*{The step where independence is used}
\begin{quote}
\textbf{Lemma.} If $\vc{u}$ and $\vc{v}$ are linearly independent, then
$D\neq0$.
\end{quote}

\noindent We prove the contrapositive: if $D=0$, then $\vc{u}$ and $\vc{v}$ are
linearly \emph{dependent}. So suppose $u_1v_2-u_2v_1=0$, and consider the two
combinations
\[
  v_1\vc{u} - u_1\vc{v} = [\,0,\;v_1u_2 - u_1v_2\,] = [\,0,\,-D\,] = \vc{0},
\]
\[
  v_2\vc{u} - u_2\vc{v} = [\,v_2u_1 - u_2v_1,\;0\,] = [\,D,\,0\,] = \vc{0},
\]
both of which vanish precisely because $D=0$. Each is a combination of $\vc{u}$
and $\vc{v}$ producing the zero vector, so we need only exhibit one whose
scalars are not both zero.

\begin{itemize}[leftmargin=1.4em,itemsep=2pt,topsep=3pt]
  \item If $v_1$ and $u_1$ are not both $0$, the first combination does it.
  \item If $v_2$ and $u_2$ are not both $0$, the second does.
  \item Otherwise all four of $u_1,u_2,v_1,v_2$ are $0$, so
        $\vc{u}=\vc{v}=\vc{0}$ --- and then $1\cdot\vc{u}+0\cdot\vc{v}=\vc{0}$
        with a nonzero scalar.
\end{itemize}

\noindent In every case $\vc{u}$ and $\vc{v}$ are dependent, which proves the
contrapositive and hence the lemma. \hfill$\blacksquare$

\subsubsection*{Finishing}
Since $\vc{u}$ and $\vc{v}$ were assumed independent, the lemma gives $D\neq0$,
so \eqref{eq:coords} defines genuine real numbers, and those numbers solve
\eqref{eq:span} --- that is, $\alpha\vc{u}+\beta\vc{v}=\vc{w}$. As $\vc{w}$ was
arbitrary,
$\R^{2}\subseteq\spanop\{\vc{u},\vc{v}\}$. Combined with Step 1,
\[
  \boxed{\;\spanop\{\vc{u},\vc{v}\} = \R^{2}\;}
\]
which is the claim. \hfill$\blacksquare$

\begin{rmk}[And the coordinates are unique]
Spanning makes $\{\vc{u},\vc{v}\}$ enough to reach everything; independence also
makes the reaching unambiguous, so the pair is a genuine basis. If
$\alpha\vc{u}+\beta\vc{v}=\alpha'\vc{u}+\beta'\vc{v}$, then rearranging gives
\[
  (\alpha-\alpha')\vc{u} + (\beta-\beta')\vc{v} = \vc{0},
\]
and independence forces both scalars to vanish, so $\alpha=\alpha'$ and
$\beta=\beta'$. Note that this argument never mentions coordinates or
components: it is the \S16.4 uniqueness proof again, but now valid for
\emph{any} independent pair rather than only for the standard basis.
\end{rmk}

\begin{ex}[The formula, checked against work already done]
For $\vc{u}=[1,2]$ and $\vc{v}=[2,3]$ we get $D = 1\cdot3 - 2\cdot2 = -1$, which
is nonzero, so the pair spans $\R^{2}$. Taking $\vc{w}=[\,1,\,0\,]$,
formula~\eqref{eq:coords} gives
\[
  \alpha = \frac{1\cdot3 - 0\cdot2}{-1} = -3,
  \qquad
  \beta  = \frac{1\cdot0 - 2\cdot1}{-1} = 2 ,
\]
which is exactly the answer we obtained by hand in \S18.1. The general formula
reproduces the specific computation, as it must.

For $\vc{u}=[2,3]$ and $\vc{v}=[3,4]$, $D = 2\cdot4 - 3\cdot3 = -1$, again
nonzero. And for the pair that failed, $\vc{u}=[1,2]$ and $\vc{v}=[2,4]$,
\[
  D = 1\cdot4 - 2\cdot2 = 0 ,
\]
so the division in \eqref{eq:coords} is unavailable --- which is the algebra
registering what \S18.2 saw geometrically, that those two vectors never leave
their line.
\end{ex}

\lookahead The quantity $D=u_1v_2-u_2v_1$ has appeared three times now, and each
time it has decided the same question: nonzero means the pair spans the plane,
zero means it collapses. A number that detects independence by a single
arithmetic test is worth having a name, and it has one. We will meet it properly
later, along with the reason the same idea keeps working in higher dimensions.

%------------------------------------------------------------------------------
\subsection{Question 2: what the dependent pair is a basis for}
%------------------------------------------------------------------------------
Return to the pair that failed,
\[
  \vc{u}=[\,1,\,2\,],
  \qquad
  \vc{v}=[\,2,\,4\,] .
\]
It does not span $\R^{2}$, but \S18.2 showed it does not span nothing either:
every combination lands on the line through the origin with slope $2$. So
$\{\vc{u},\vc{v}\}$ is doing \emph{some} job. The question is what job, and with
how many vectors.

\subsubsection*{One of the two is redundant}
Because $\vc{v}=2\vc{u}$, the second vector contributes nothing the first did not
already contribute. The computation of \S18.2 says it exactly:
\[
  \alpha\vc{u} + \beta\vc{v}
  = \alpha\vc{u} + \beta(2\vc{u})
  = (\alpha+2\beta)\,\vc{u} ,
\]
so every combination of the two is already a multiple of $\vc{u}$ alone. That
gives $\spanop\{\vc{u},\vc{v}\}\subseteq\spanop\{\vc{u}\}$; and the reverse
inclusion is immediate, since $\alpha\vc{u}=\alpha\vc{u}+0\vc{v}$. Hence
\[
  \spanop\{\vc{u},\vc{v}\} \;=\; \spanop\{\vc{u}\} .
\]
Dropping $\vc{v}$ costs us nothing at all. So keep one of them --- take
$\vc{u}=[\,1,\,2\,]$ --- and write
\[
  \bas = \bigl\{\,[\,1,\,2\,]\,\bigr\} .
\]

\subsubsection*{What that one vector spans}
By definition the span of a single vector is the set of its scalar multiples:
\[
  L \;=\; \spanop\bas
    \;=\; \bigl\{\, \alpha[\,1,\,2\,] \;\bigm|\; \alpha\in\R \,\bigr\}
    \;=\; \bigl\{\,(x,y)\in\R^{2} \;\bigm|\; y=2x \,\bigr\} .
\]
Every vector in $L$ is a scalar multiple of $[\,1,\,2\,]$, and every scalar
multiple of $[\,1,\,2\,]$ is in $L$. Geometrically $L$ is the line through the
origin in the direction of $[\,1,\,2\,]$ --- the dotted line of \S18.2, now
named.

\begin{center}
\begin{tikzpicture}[scale=0.85]
  \lattice{-3}{4}{-3}{5}
  \draw[gray!65,dotted,thick] (-1.55,-3.1) -- (2.55,5.1);
  \draw[vecarr,RoyalBlue] (0,0) -- (1,2);
  \fill[ForestGreen] (2,4)    circle (2.1pt);
  \fill[ForestGreen] (0.5,1)  circle (2.1pt);
  \fill[ForestGreen] (-1,-2)  circle (2.1pt);
  \node[left,font=\scriptsize,RoyalBlue]   at (0.72,2.10) {$\vc{u}=[1,2]$};
  \node[right,font=\scriptsize,ForestGreen] at (2.08,4.02) {$2\vc{u}$};
  \node[right,font=\scriptsize,ForestGreen] at (0.58,1.0)  {$\tfrac12\vc{u}$};
  \node[right,font=\scriptsize,ForestGreen] at (-0.9,-2.0) {$-\vc{u}$};
  \node[right,font=\scriptsize,gray]        at (2.5,4.92)  {$L=\spanop\bas$};
\end{tikzpicture}
\end{center}

\subsubsection*{And it is a basis for that line}
$\bas$ spans $L$ by construction. It also has no redundancy, which for a
one-element set is the statement that the vector is linearly independent on its
own: if $\alpha[\,1,\,2\,]=\vc{0}$ then $\alpha\cdot1=0$, so $\alpha=0$, and the
only combination giving the zero vector is the trivial one. So
\[
  \bas=\bigl\{[\,1,\,2\,]\bigr\}
  \quad\text{is a basis for}\quad L ,
  \qquad\text{and}\qquad
  \dim L = 1 ,
\]
since a basis for $L$ has one vector in it. One basis vector, one independent
direction, one free number $\alpha$ needed to say where on the line you are ---
the three counts agree, exactly as \S17 said they must.

\smallskip
\noindent That answers \textbf{Q2}. The pair $[\,1,\,2\,]$ and $[\,2,\,4\,]$ is
not a basis for anything: it is dependent, so it fails the no-redundancy
requirement wherever you point it. What it is, is a \emph{spanning set} for the
line $L$ --- enough to reach every point of $L$, but with one vector more than
necessary. Discard the surplus and $\{[\,1,\,2\,]\}$ is a basis for $L$, a
one-dimensional line sitting inside the two-dimensional plane.

\begin{rmk}[Spanning set and basis are not the same thing]
This is the distinction the failed pair exposes. A \textbf{spanning set} reaches
everything; a \textbf{basis} reaches everything \emph{and} carries no
passengers. Every basis is a spanning set, but not conversely --- and the gap
between the two is exactly linear dependence. Compare \S16.3, where adding
$[\,3,\,5\,]$ to $\{\ivec,\jvec\}$ left a spanning set that had stopped being a
basis.
\end{rmk}

\begin{rmk}[The choice of which vector to keep is arbitrary]
Nothing forced us to keep $[\,1,\,2\,]$ rather than $[\,2,\,4\,]$. The set
$\bas'=\{[\,2,\,4\,]\}$ spans the same line, because
$\alpha[\,2,\,4\,]=2\alpha[\,1,\,2\,]$ and $2\alpha$ runs over all of $\R$ as
$\alpha$ does. So $\bas$ and $\bas'$ are different bases for the same line $L$.
A space does not have \emph{the} basis; it has many, and we choose whichever is
convenient. Hold on to that observation --- it is most of the answer to
\textbf{Q3}.
\end{rmk}

\begin{rmk}[Every span contains the origin]
Taking all scalars to be $0$ gives the zero vector, so $\vc{0}$ lies in the span
of any collection whatsoever. In particular $L$ passes through the origin, and a
line that misses the origin --- $y=2x+1$, say --- is not the span of anything.
Spans are anchored at $\vc{0}$.
\end{rmk}

\noindent Note the pattern the two questions have produced. An independent pair
gives a basis of dimension $2$ and spans the whole plane; a dependent pair
collapses to dimension $1$ and spans a line. In both cases what we end up
holding is a set with its own basis and its own dimension, sitting inside
$\R^{2}$. That object is worth isolating, and \S19 does so. \textbf{Q3} remains.

%==============================================================================
\section{Subspaces}
%==============================================================================
We have now met two rather different things and treated them the same way. One
is $\R^{2}$, with basis $\{\ivec,\jvec\}$ and dimension $2$. The other is the
line $L=\spanop\{[1,2]\}$, with basis $\{[1,2]\}$ and dimension $1$. The second
sits inside the first, and yet everything we did with $\R^{2}$ --- pick a basis,
count it, write vectors in coordinates --- worked verbatim on $L$.

That is suspicious in a productive way. If a proper subset of $\R^{2}$ supports
the whole apparatus, then the apparatus was never really about $\R^{2}$; it was
about some property that $\R^{2}$ and $L$ share. This section finds the
property.

%------------------------------------------------------------------------------
\subsection{What \texorpdfstring{$\R^{2}$}{R2} and the line have in common}
%------------------------------------------------------------------------------
Ask what you are allowed to \emph{do} inside each. In $\R^{2}$ you may add two
vectors, or scale one, and the result is again in $\R^{2}$. The same is true in
$L$: adding two multiples of $[1,2]$ gives another multiple of $[1,2]$, and
scaling a multiple of $[1,2]$ gives another one. Neither operation can carry you
out of the set you started in.

Call that property \textbf{closure}. It is easy to state and easy to fail, and
the failures are the fastest way to see that it is saying something.

\begin{ex}[A line that misses the origin]
Let $S=\{(x,y)\mid y=2x+1\}$ --- the line of $L$, shifted up by one. The points
$(0,1)$ and $(1,3)$ both lie on it. Their sum is $(1,4)$, and
$2\cdot1+1=3\neq4$, so the sum has left $S$. Closure under addition fails.
Notice also that $\vc{0}=(0,0)$ is not in $S$, since $2\cdot0+1=1\neq0$.
\end{ex}

\begin{ex}[The first quadrant]
Let $S=\{(x,y)\mid x\ge0 \text{ and } y\ge0\}$. Adding two vectors with
non-negative entries gives a vector with non-negative entries, so closure under
\emph{addition} holds. But scaling breaks it: $[\,2,\,1\,]$ is in $S$ and
$(-1)[\,2,\,1\,]=[\,-2,\,-1\,]$ is not. One of the two operations is enough to
disqualify it.
\end{ex}

  \lattice{-1}{3}{-1}{5}
  \draw[BrickRed,dashed,thick] (-1,-1) -- (2.1,5.2);
  \fill[RoyalBlue] (0,1) circle (2.2pt);
  \fill[RoyalBlue] (1,3) circle (2.2pt);
  \fill[BrickRed]  (1,4) circle (2.2pt);

  \node[right,font=\scriptsize,RoyalBlue] at (1.08,2.9) {$(1,3)$};
  \node[right,font=\scriptsize,BrickRed]  at (1.08,4.1) {$(1,4)$ --- off};
  \node[right,font=\scriptsize,BrickRed]  at (2.0,5.05) {$y=2x+1$};
\end{tikzpicture}

%------------------------------------------------------------------------------
\subsection{Spans are automatically closed}
%------------------------------------------------------------------------------
Here is why the property is worth naming rather than merely noticing: it is not
a coincidence that $\R^{2}$ and $L$ both have it. \emph{Every} span has it, and
the check is three lines.

Take any $\vc{u},\vc{v}\in\R^{2}$ and let $S=\spanop\{\vc{u},\vc{v}\}$. Two
elements of $S$ look like
\[
  \vc{w}_1 = \alpha_1\vc{u}+\beta_1\vc{v},
  \qquad
  \vc{w}_2 = \alpha_2\vc{u}+\beta_2\vc{v} .
\]
Adding them and collecting terms,
\[
  \vc{w}_1+\vc{w}_2 = (\alpha_1+\alpha_2)\vc{u} + (\beta_1+\beta_2)\vc{v} ,
\]
which is again a linear combination of $\vc{u}$ and $\vc{v}$, hence again in
$S$. Scaling by $\gamma\in\R$,
\[
  \gamma\vc{w}_1 = (\gamma\alpha_1)\vc{u} + (\gamma\beta_1)\vc{v} ,
\]
likewise in $S$. Nothing about the argument used the number of vectors, so the
same holds for $\spanop\{\vc{u}\}$ --- which is how $L$ came by the property ---
and would hold for a span of ten vectors.

So spans are closed, always. And that suggests the definition should be built
around closure.

%------------------------------------------------------------------------------
\subsection{The definition}
%------------------------------------------------------------------------------
\begin{defn}[Subspace of $\R^{2}$]
A subset $S\subseteq\R^{2}$ is a \textbf{subspace} of $\R^{2}$ when
\begin{enumerate}[label=\textbf{S\arabic*.},leftmargin=3.0em,itemsep=3pt,
                  topsep=3pt]
  \item $S$ is non-empty;
  \item $S$ is closed under addition: if $\vc{w}_1,\vc{w}_2\in S$ then
        $\vc{w}_1+\vc{w}_2\in S$;
  \item $S$ is closed under scalar multiplication: if $\vc{w}\in S$ and
        $\gamma\in\R$ then $\gamma\vc{w}\in S$.
\end{enumerate}
\end{defn}

\begin{rmk}[Every subspace contains the origin]
$\vc{0}$ is never mentioned in the definition, and never needs to be. Since $S$
is non-empty, pick any $\vc{w}\in S$; then \textbf{S3} with $\gamma=0$ puts
$0\vc{w}=\vc{0}$ in $S$. So the origin is forced, which matches the remark in
\S18.6 that every span contains $\vc{0}$ --- and it is precisely what the
shifted line $y=2x+1$ failed. In practice this is the fastest disqualifier
available: if a set misses the origin, stop, it is not a subspace.
\end{rmk}

\noindent By \S19.2, $\spanop\{\vc{u},\vc{v}\}$ and $\spanop\{\vc{u}\}$ are
subspaces, so both objects we started with qualify: $\R^{2}$ is a subspace of
itself, and the line $L$ is a subspace of $\R^{2}$. The apparatus of bases,
coordinates and dimension was never about $\R^{2}$ specifically --- it works for
any subspace, which is why it worked on $L$ without modification.

%------------------------------------------------------------------------------
\subsection{All of them}
%------------------------------------------------------------------------------
In the plane the definition is restrictive enough that we can list every
subspace there is, and the argument is one we have already done all the work
for.

\begin{quote}
\textbf{Claim.} The subspaces of $\R^{2}$ are exactly: the single point
$\{\vc{0}\}$; every line through the origin; and $\R^{2}$ itself.
\end{quote}

\noindent Let $S$ be a subspace. Either it contains a nonzero vector or it does
not.

\begin{itemize}[leftmargin=1.4em,itemsep=4pt]
  \item If it does not, then $S=\{\vc{0}\}$, since the origin is forced to be
        there and nothing else is.
  \item Otherwise pick $\vc{u}\in S$ with $\vc{u}\neq\vc{0}$. Closure under
        scaling puts every multiple of $\vc{u}$ in $S$, so
        $\spanop\{\vc{u}\}\subseteq S$: the whole line through $\vc{u}$ is
        already inside. If that is all of $S$, then $S$ is a line through the
        origin.
  \item If not, there is some $\vc{v}\in S$ lying off that line. Then $\vc{u}$
        and $\vc{v}$ are linearly independent --- $\vc{v}$ is nonzero, and it is
        not a multiple of $\vc{u}$, which by \S18.4 is exactly independence for
        a pair. So \S18.5 gives $\spanop\{\vc{u},\vc{v}\}=\R^{2}$; and closure
        keeps every one of those combinations inside $S$. Hence $S=\R^{2}$.
\end{itemize}

\noindent That shows every subspace is one of the three. The converse needs no
new work: each of the three is a span, and spans are subspaces by \S19.2 ---
$\{\vc{0}\}=\spanop\{\vc{0}\}$, a line through the origin is $\spanop\{\vc{u}\}$
for any $\vc{u}$ on it, and $\R^{2}=\spanop\{\ivec,\jvec\}$. So the list is
complete in both directions, which is what ``exactly'' claimed.

Reading off the dimensions, the three cases have dimension $0$, $1$
and $2$ --- nothing in between, and nothing above. A subspace of the plane
cannot have dimension $\tfrac32$, and it cannot have dimension $3$.

\begin{rmk}
The dimension of $\{\vc{0}\}$ is $0$ by the same counting rule as everywhere
else: a basis is a spanning set with no redundancy, and $\{\vc{0}\}$ is spanned
by no vectors at all --- the empty collection --- because the one vector it
contains would be redundant. A basis with nothing in it has size $0$.
\end{rmk}

\lookahead Notice what the classification did \emph{not} use: any picture. It
used closure, the definition of independence, and the spanning theorem of
\S18.5. That is the pattern to expect --- as the dimension grows the pictures
give out long before the arguments do, and in $\R^{3}$ the subspaces will turn
out to be the origin, the lines through it, the planes through it, and the whole
space, proved the same way.

%==============================================================================
\section{Stepping up: \texorpdfstring{$\R^{3}$}{R3}}
%==============================================================================
Return to the standard basis $\{\ivec,\jvec\}$ and ask a new question. What
happens if we introduce a genuinely new independent direction --- one that no
combination of $\ivec$ and $\jvec$ can produce --- and lengthen every vector by a
third coordinate to make room for it?

%------------------------------------------------------------------------------
\subsection{A third direction}
%------------------------------------------------------------------------------
\begin{defn}[The standard basis of $\R^{3}$]
Write
\[
  \ivec = [\,1,\,0,\,0\,],
  \qquad
  \jvec = [\,0,\,1,\,0\,],
  \qquad
  \kvec = [\,0,\,0,\,1\,] ,
\]
vectors in
$\R^{3}=\{(x,y,z)\mid x,y,z\in\R\}$. The set $\{\ivec,\jvec,\kvec\}$ is the
\textbf{standard basis} of $\R^{3}$.
\end{defn}

\noindent Everything \S16 said carries over with a third slot appended. A linear
combination is
\[
  \alpha\ivec + \beta\jvec + \gamma\kvec = [\,\alpha,\,\beta,\,\gamma\,] ,
\]
so any choice of three scalars lands in $\R^{3}$; and conversely any
$\vc{v}=[\,v_1,\,v_2,\,v_3\,]$ is reached by taking $\alpha=v_1$, $\beta=v_2$,
$\gamma=v_3$. Hence
\[
  \spanop\{\ivec,\jvec,\kvec\} = \R^{3},
  \qquad
  \dim\R^{3} = 3 .
\]
Three basis vectors, three independent directions, three free numbers. The
counting rule of \S17 has not changed; only the count has.

The three are still orthonormal, by the same one-line checks as before: each has
norm $1$, and every pair has dot product zero, since in each product every term
has a $0$ in it. In particular $\langle\ivec,\kvec\rangle=0$ and
$\langle\jvec,\kvec\rangle=0$.

%------------------------------------------------------------------------------
\subsection{The floor, and a subspace we already know}
%------------------------------------------------------------------------------
Picture $\ivec$ and $\jvec$ as lying flat --- they span a horizontal plane, the
\emph{floor} --- and $\kvec$ as pointing straight up out of it.

\begin{center}
\begin{tikzpicture}[scale=1.35]
  % the floor: the plane spanned by i and j, drawn in projection
  \fill[RoyalBlue!10] (0,0) -- (-1.1,-0.8) -- (0.9,-0.8) -- (2,0) -- cycle;
  \draw[RoyalBlue!45] (0,0) -- (-1.1,-0.8) -- (0.9,-0.8) -- (2,0) -- cycle;
  \draw[axisline] (0,0) -- (-1.35,-0.98) node[left,font=\scriptsize,black]{$x$};
  \draw[axisline] (0,0) -- (2.55,0) node[right,font=\scriptsize,black]{$y$};
  \draw[axisline] (0,0) -- (0,1.95) node[above,font=\scriptsize,black]{$z$};
  \draw[vecarr,RoyalBlue]   (0,0) -- (-0.55,-0.4);
  \draw[vecarr,ForestGreen] (0,0) -- (1,0);
  \draw[vecarr,BrickRed]    (0,0) -- (0,1);
  \node[below,font=\small,RoyalBlue]        at (-0.60,-0.40) {$\ivec$};
  \node[above,font=\small,ForestGreen]      at (1,0.03)      {$\jvec$};
  \node[right,font=\small,BrickRed]         at (0.05,1)      {$\kvec$};
  \node[font=\scriptsize,RoyalBlue!75]      at (1.15,-0.55)
        {the floor: $\spanop\{\ivec,\jvec\}$};
\end{tikzpicture}
\end{center}

\noindent That floor is not a new kind of object. It is
$\spanop\{\ivec,\jvec\}$, so by \S19.2 it is a \emph{subspace} of $\R^{3}$ ---
closed under addition and scaling, containing the origin, with basis
$\{\ivec,\jvec\}$ and dimension $2$. A two-dimensional subspace sitting inside a
three-dimensional space, exactly as the line $L$ of \S18.6 was a
one-dimensional subspace sitting inside the plane.

And $\kvec$ is what makes the third dimension real: it lies off the floor, so it
is not in $\spanop\{\ivec,\jvec\}$, and adjoining it is what takes us from a
subspace of dimension $2$ up to the whole of $\R^{3}$. That is the same move as
in the classification of \S19.4 --- find a vector outside what you have, and the
span grows.

%------------------------------------------------------------------------------
\subsection{Everything generalizes}
%------------------------------------------------------------------------------
This is the point of the whole unit, and it is worth stating plainly: almost
nothing we built in $\R^{2}$ was really about $\R^{2}$. The definitions were
written in terms of coordinates and they extend by appending a term. Some
formulas do not even change, because they were written in terms of the ones that
did.

\begin{center}
\renewcommand{\arraystretch}{1.7}
\begin{tabular}{@{}l l l@{}}
\toprule
\textbf{Object} & \textbf{in $\R^{2}$} & \textbf{in $\R^{3}$} \\
\midrule
Norm (\S2)
  & $\sqrt{v_1^{2}+v_2^{2}}$
  & $\sqrt{v_1^{2}+v_2^{2}+v_3^{2}}$ \\
Dot product (\S11.3)
  & $u_1v_1+u_2v_2$
  & $u_1v_1+u_2v_2+u_3v_3$ \\
$p$-norm (\S14.6)
  & $\bigl(\lvert v_1\rvert^{p}+\lvert v_2\rvert^{p}\bigr)^{1/p}$
  & $\bigl(\lvert v_1\rvert^{p}+\lvert v_2\rvert^{p}+\lvert v_3\rvert^{p}\bigr)^{1/p}$ \\
\midrule
Cosine similarity (\S11.4)
  & $\dfrac{\langle\vc{u},\vc{v}\rangle}{\norm{\vc{u}}\norm{\vc{v}}}$
  & \emph{unchanged} \\
Projection (\S12.4)
  & $\dfrac{\langle\vc{u},\vc{v}\rangle}{\norm{\vc{u}}^{2}}\,\vc{u}$
  & \emph{unchanged} \\
Metric (\S14.2)
  & $\norm{\vc{x}-\vc{y}}$
  & \emph{unchanged} \\
\bottomrule
\end{tabular}
\end{center}

\noindent The split in that table is the whole story. The top three are defined
directly from coordinates, so each gains a term. The bottom three are defined
from the top three, so they are \emph{character for character identical} --- the
projection formula does not know or care how many coordinates a vector has.
Every theorem proved from those formulas transfers as well: Cauchy--Schwarz, the
triangle inequality, the orthogonality test, the fact that
$\langle\vc{u},\vc{u}\rangle=\norm{\vc{u}}^{2}$.

\subsubsection*{Worked examples}
Throughout, take
\[
  \vc{u} = [\,1,\,2,\,2\,],
  \qquad
  \vc{v} = [\,2,\,2,\,1\,],
  \qquad
  \vc{w} = [\,2,\,-2,\,1\,] .
\]

\begin{ex}[Operations]
$\vc{u}+\vc{v}=[\,3,\,4,\,3\,]$ and $\vc{u}-\vc{v}=[\,-1,\,0,\,1\,]$, both
componentwise as in \S7. Scaling, $3\vc{u}=[\,3,\,6,\,6\,]$. In basis form,
$\vc{u}=\ivec+2\jvec+2\kvec$.
\end{ex}

\begin{ex}[Norms]
$\norm{\vc{u}}=\sqrt{1+4+4}=\sqrt9=3$, and likewise $\norm{\vc{v}}=3$ and
$\norm{\vc{w}}=3$. The other members of the family behave as \S14 promised:
\[
  \norm{\vc{u}}_1 = 1+2+2 = 5,
  \qquad
  \norm{\vc{u}}_2 = 3,
  \qquad
  \norm{\vc{u}}_\infty = \max\{1,2,2\} = 2 ,
\]
and indeed $2\le3\le5$, the ordering
$\norm{\vc{u}}_\infty\le\norm{\vc{u}}_2\le\norm{\vc{u}}_1$ of \S14.8. Homogeneity
also survives: $\norm{3\vc{u}}=\sqrt{9+36+36}=9=3\norm{\vc{u}}$.
\end{ex}

\begin{ex}[Dot product, cosine similarity, orthogonality]
$\langle\vc{u},\vc{v}\rangle = 1(2)+2(2)+2(1) = 8$, so
\[
  \cos(\vc{u},\vc{v}) = \frac{8}{3\cdot3} = \frac{8}{9}\approx0.889 ,
\]
close to $1$: the two point in nearly the same direction. Cauchy--Schwarz is
comfortable, $8\le9$. Meanwhile
\[
  \langle\vc{u},\vc{w}\rangle = 1(2)+2(-2)+2(1) = 2-4+2 = 0 ,
\]
so by the test of \S11.7, $\vc{u}$ and $\vc{w}$ are \emph{orthogonal} --- a
right angle in space, detected by three multiplications and two additions, with
no picture consulted.
\end{ex}

\begin{ex}[Projection]
Project $\vc{v}$ onto $\vc{u}$. Since $\langle\vc{u},\vc{v}\rangle=8$ and
$\norm{\vc{u}}^{2}=9$,
\[
  \comp{\vc{u}}{\vc{v}} = \frac{8}{3},
  \qquad
  \proj{\vc{u}}{\vc{v}} = \frac{8}{9}[\,1,\,2,\,2\,]
    = \Bigl[\,\tfrac89,\ \tfrac{16}{9},\ \tfrac{16}{9}\,\Bigr] .
\]
The orthogonal piece of \S12.6 still works:
\[
  \vc{v}-\proj{\vc{u}}{\vc{v}}
  = \Bigl[\,\tfrac{10}{9},\ \tfrac{2}{9},\ -\tfrac79\,\Bigr],
  \qquad
  \bigl\langle\vc{u},\ \vc{v}-\proj{\vc{u}}{\vc{v}}\bigr\rangle
  = \tfrac{10}{9}+\tfrac49-\tfrac{14}{9} = 0 . \quad\checkmark
\]
\end{ex}

\begin{ex}[Metrics]
The distance from $\vc{u}$ to $\vc{v}$ is
$\norm{\vc{u}-\vc{v}}=\norm{[-1,0,1]}=\sqrt2$. Under the other two metrics of
\S14,
\[
  d_1(\vc{u},\vc{v}) = 1+0+1 = 2,
  \qquad
  d_\infty(\vc{u},\vc{v}) = \max\{1,0,1\} = 1 ,
\]
and $1\le\sqrt2\le2$, the same ordering as before.
\end{ex}

%------------------------------------------------------------------------------
\subsection{What does change}
%------------------------------------------------------------------------------
Two honest qualifications.

First, the results that counted to two now count to three. A basis for $\R^{3}$
needs \emph{three} independent vectors, not two; two independent vectors in
$\R^{3}$ span only a plane through the origin. And the test $D=u_1v_2-u_2v_1$ of
\S18.5 was built for a $2\times2$ system and does not apply as written.

Second, $\R^{3}$ supports an operation with no analogue in $\R^{2}$: the
\textbf{cross product}, which takes two vectors and returns a third vector
perpendicular to both. This is the $\times$ the complex numbers lecture warned
against confusing with the dot product --- and now it is clear why the warning
was specific to $\R^{3}$.

\lookahead Nothing above stopped at three. The same definitions read perfectly
well in $\R^{n}$ with $n$ terms in each sum, and that is how we will use them.
The one thing genuinely lost is the picture: in $\R^{2}$ and $\R^{3}$ you can see
the angle between two vectors, and past that you cannot. This is exactly the
situation the remark in \S15.3 anticipated --- in high dimensions the angle is
\emph{defined} by $\cos(\vc{u},\vc{v})=\langle\vc{u},\vc{v}\rangle/(\norm{\vc{u}}\norm{\vc{v}})$,
and Cauchy--Schwarz is what guarantees that quotient is a legal cosine so the
definition makes sense. The algebra outlives the geometry, which is why we were
careful to prove things rather than draw them.

%------------------------------------------------------------------------------
\subsection{Practice}
%------------------------------------------------------------------------------
Let
\[
  \vc{a} = [\,1,\,2,\,2\,],
  \qquad
  \vc{b} = [\,2,\,-2,\,1\,],
  \qquad
  \vc{c} = [\,3,\,0,\,4\,] .
\]

\begin{enumerate}[label=\textbf{P\arabic*.},leftmargin=2.8em,itemsep=6pt,
                  topsep=6pt]
  \item Compute $\vc{a}+\vc{b}$, \ $\vc{a}-\vc{c}$, \ and $2\vc{b}$.
  \item Compute $\norm{\vc{a}}$, $\norm{\vc{b}}$ and $\norm{\vc{c}}$.
  \item Compute $\langle\vc{a},\vc{b}\rangle$. Are $\vc{a}$ and $\vc{b}$
        orthogonal?
  \item Compute $\cos(\vc{b},\vc{c})$.
  \item Compute $\comp{\vc{a}}{\vc{c}}$ and $\proj{\vc{a}}{\vc{c}}$. Then verify
        that $\vc{c}-\proj{\vc{a}}{\vc{c}}$ is orthogonal to $\vc{a}$.
  \item Compute the distance from $\vc{a}$ to $\vc{c}$ under the Euclidean,
        Manhattan and Chebyshev metrics, and check that they come out in the
        order \S14.8 predicts.
  \item Write $\vc{c}$ as a linear combination of $\ivec$, $\jvec$ and $\kvec$.
\end{enumerate}

\subsubsection*{Answer key}
\begin{enumerate}[label=\textbf{P\arabic*.},leftmargin=2.8em,itemsep=5pt,
                  topsep=5pt]
  \item $\vc{a}+\vc{b}=[\,3,\,0,\,3\,]$, \
        $\vc{a}-\vc{c}=[\,-2,\,2,\,-2\,]$, \
        $2\vc{b}=[\,4,\,-4,\,2\,]$.
  \item $\norm{\vc{a}}=\sqrt{1+4+4}=3$, \
        $\norm{\vc{b}}=\sqrt{4+4+1}=3$, \
        $\norm{\vc{c}}=\sqrt{9+0+16}=5$.
  \item $\langle\vc{a},\vc{b}\rangle = 2-4+2 = 0$, so yes --- orthogonal.
  \item $\langle\vc{b},\vc{c}\rangle = 6+0+4 = 10$, so
        $\cos(\vc{b},\vc{c}) = \dfrac{10}{3\cdot5} = \dfrac23$.
  \item $\langle\vc{a},\vc{c}\rangle = 3+0+8 = 11$ and $\norm{\vc{a}}^{2}=9$, so
        $\comp{\vc{a}}{\vc{c}} = \tfrac{11}{3}$ and
        $\proj{\vc{a}}{\vc{c}} = \tfrac{11}{9}[\,1,\,2,\,2\,]
         = \bigl[\,\tfrac{11}{9},\,\tfrac{22}{9},\,\tfrac{22}{9}\,\bigr]$.
        Then $\vc{c}-\proj{\vc{a}}{\vc{c}}
         = \bigl[\,\tfrac{16}{9},\,-\tfrac{22}{9},\,\tfrac{14}{9}\,\bigr]$, and
        $\tfrac{16}{9}-\tfrac{44}{9}+\tfrac{28}{9}=0$. \checkmark
  \item $\vc{a}-\vc{c}=[\,-2,\,2,\,-2\,]$, so $d_2=\sqrt{12}=2\sqrt3\approx3.46$,
        $d_1=6$, and $d_\infty=2$. Indeed $2\le2\sqrt3\le6$.
  \item $\vc{c} = 3\ivec + 0\jvec + 4\kvec = 3\ivec+4\kvec$.
\end{enumerate}
